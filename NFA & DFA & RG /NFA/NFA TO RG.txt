#include <iostream>
#include <vector>
#include <unordered_set>
#include <unordered_map>
#include <queue>

using namespace std;

class NFA {
private:
    int numStates;
    unordered_map<char, vector<int>> transitions;
    unordered_set<int> acceptStates;

public:
    NFA(int states) : numStates(states) {}

    void addTransition(int from, char input, int to) {
        transitions[input].push_back(to);
    }

    void addAcceptState(int state) {
        acceptStates.insert(state);
    }

    string convertToRegularGrammar() {
        unordered_map<int, unordered_set<string>> productions;

        // Initialize with epsilon productions
        for (int state = 0; state < numStates; ++state) {
            if (acceptStates.find(state) != acceptStates.end()) {
                productions[state].insert("ε");
            }
        }

        // BFS to generate productions
        queue<int> q;
        q.push(0); // Start state

        while (!q.empty()) {
            int state = q.front();
            q.pop();

            for (const auto& transition : transitions) {
                char input = transition.first;
                for (int nextState : transition.second) {
                    if (productions[nextState].empty()) {
                        productions[nextState].insert(input + "A" + to_string(nextState));
                        q.push(nextState);
                    }
                }
            }
        }

        // Construct the regular grammar
        string regularGrammar;
        for (const auto& production : productions) {
            int state = production.first;
            for (const string& rule : production.second) {
                regularGrammar += "A" + to_string(state) + " -> " + rule + " | ";
            }
            regularGrammar.pop_back(); // Remove the last '|'
            regularGrammar.pop_back(); // Remove the space before '|'
            regularGrammar += "\n";
        }

        return regularGrammar;
    }
};

int main() {
    int numStates, numTransitions, numAcceptStates;
    cout << "Enter the number of states: ";
    cin >> numStates;

    NFA nfa(numStates);

    cout << "Enter the number of transitions: ";
    cin >> numTransitions;

    cout << "Enter transitions in the format (fromState inputSymbol toState):" << endl;
    for (int i = 0; i < numTransitions; ++i) {
        int from, to;
        char input;
        cin >> from >> input >> to;
        nfa.addTransition(from, input, to);
    }

    cout << "Enter the number of accepting states: ";
    cin >> numAcceptStates;

    cout << "Enter the accepting states: ";
    for (int i = 0; i < numAcceptStates; ++i) {
        int state;
        cin >> state;
        nfa.addAcceptState(state);
    }

    // Convert to regular grammar
    string regularGrammar = nfa.convertToRegularGrammar();
    cout << "Regular Grammar:\n" << regularGrammar;

    return 0;
}
