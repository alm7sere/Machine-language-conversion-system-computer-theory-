class NFA:
    def __init__(self, states, alphabet, transitions, start_state, accept_states):
        self.states = states
        self.alphabet = alphabet
        self.transitions = transitions
        self.start_state = start_state
        self.accept_states = set(accept_states)  # Convert to set

    def epsilon_closure(self, state, visited=None):
        if visited is None:
            visited = set()
        visited.add(state)
        closure = set()
        closure.add(state)
        transitions = self.transitions.get((state, '@'), [])
        for next_state in transitions:
            if next_state not in visited:
                closure.update(self.epsilon_closure(next_state, visited))
        return closure

    def move(self, states, symbol):
        result = set()
        for state in states:
            transitions = self.transitions.get((state, symbol), [])
            for next_state in transitions:
                result.update(self.epsilon_closure(next_state))
        return result

    def convert_to_dfa(self):
        dfa_transitions = {}
        dfa_start_state = frozenset(self.epsilon_closure(self.start_state))
        dfa_states = {dfa_start_state}
        dfa_accept_states = set()
        queue = [dfa_start_state]

        while queue:
            current_state = queue.pop(0)
            for symbol in self.alphabet:
                next_state = frozenset(self.move(current_state, symbol))
                if next_state:
                    dfa_transitions[(current_state, symbol)] = next_state
                    if next_state not in dfa_states:
                        dfa_states.add(next_state)
                        queue.append(next_state)
                    if self.accept_states.intersection(set(next_state)):
                        dfa_accept_states.add(next_state)

        return DFA(dfa_states, self.alphabet, dfa_transitions, dfa_start_state, dfa_accept_states)

    def print_table(self, name):
        print(name + ":")
        for state in sorted(self.states, key=lambda s: sorted(list(s))):
            if state == '':
                continue
            for symbol in self.alphabet:
                next_state = self.transitions.get((state, symbol), [])
                next_state = next_state if next_state else []
                if '@' in self.alphabet:
                    lambda_transitions = self.transitions.get((state, '@'), [])
                    next_state += lambda_transitions
                if next_state:
                    print("{}==={}===>{}".format(state, symbol, next_state))
        print("Accept states in {}: {}".format(name, self.accept_states))

        transitions[(from_state.strip(), symbol.strip())] = to_states

    return states, alphabet, transitions, start_state, accept_states
